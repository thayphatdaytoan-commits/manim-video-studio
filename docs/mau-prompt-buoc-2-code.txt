=== ĐẦU FILE (COPY NGUYÊN KHỐI) ===
from manim import *

config.pixel_width = 1080
config.pixel_height = 1920
TOP_BUFF = 0.05
BOTTOM_BUFF = 0.05
MARGIN = 0.08
FIGURE_RATIO = 0.58
SAFE_W = config.frame_width - 2 * MARGIN
MAX_LINES_PER_PAGE = 4

STYLE_VN = {"bg": "#0d1117", "point": "#8b1a1a", "segment": "#1e40af",
            "circle": "#3d6b2f", "text": "#FFFFFF", "highlight": "#FFD700", "conclusion": "#FF8C00"}

def center_x(mob):
    mob.set_x(0)
    return mob

def vn(s, size=30, color=None):
    return Text(s, font="Arial", font_size=size,
                color=color or STYLE_VN["text"], disable_ligatures=True)

def fit_figure_full_width(fig, max_h):
    fig.scale_to_fit_width(SAFE_W)
    if fig.height > max_h:
        fig.scale_to_fit_height(max_h)
    return fig

def interior_angle_at(vertex, arm1, arm2, radius=0.28, color=None, **kwargs):
  v = vertex.get_center(); a1 = arm1.get_center(); a2 = arm2.get_center()
  return Angle(Line(v,a1,buff=0), Line(v,a2,buff=0), radius=radius, other_angle=False,
               color=color or STYLE_VN["highlight"], **kwargs)

def right_angle_at(vertex, arm1, arm2, length=0.22, **kwargs):
  v = vertex.get_center(); a1 = arm1.get_center(); a2 = arm2.get_center()
  return RightAngle(Line(v,a1,buff=0), Line(v,a2,buff=0), length=length, **kwargs)

=== LUỒNG construct() ===
1. problem_block.to_edge(UP, buff=TOP_BUFF); center_x(problem_block)
2. figure = VGroup(...); fit_figure_full_width(figure, avail_h); next_to(problem_block, DOWN, buff=0.1); center_x(figure)
3. FadeOut(problem_block); fit_figure_full_width(figure, frame_height*FIGURE_RATIO); to_edge(UP, buff=TOP_BUFF); center_x(figure)
4. solution_stack: mỗi dòng next_to(figure/stack, DOWN, buff=0.08) → center_x(dòng)
5. page_break sau 4 dòng: FadeOut(solution_stack) — GIỮ figure

=== CẤM ===
align_to(LEFT_EDGE, LEFT), Tex(), move_to(LEFT*2.8), to_edge(RIGHT), scale(0.38), shift(UP*2), font≤24, TOP_BUFF>0.12
Angle/Arc reflex >180° — dùng interior_angle_at(đỉnh, arm1, arm2); CẤM other_angle=True
